    BEFORE INSERT OR UPDATE 
    ON GS_USER
    FOR EACH ROW

END GS_USER_T;
/

create or replace function GS_GET_CURRENT_STOCK(P_TYPE_ID NUMBER,P_GS_COMPANY_ID NUMBER,	P_GS_BARANCH_ID NUMBER)
RETURN NUMBER
IS 
V_STOCK NUMBER;
BEGIN
    SELECT NVL(GS_GET_PURCHASE_QTY(P_TYPE_ID,P_GS_COMPANY_ID,P_GS_BARANCH_ID),0)-NVL(GS_GET_SALE_QTY(P_TYPE_ID,P_GS_COMPANY_ID,P_GS_BARANCH_ID),0) INTO V_STOCK FROM DUAL;
    RETURN V_STOCK;
END;
/
create or replace function GS_GET_CURRENT_STOCK_DATE_WISE(P_TYPE_ID NUMBER,P_GS_COMPANY_ID NUMBER,	P_GS_BARANCH_ID NUMBER,P_FROM DATE,P_TO DATE)
RETURN NUMBER
IS 
V_STOCK NUMBER;
BEGIN
    SELECT NVL(GS_GET_PURCHASE_QTY_DATE_WISE(P_TYPE_ID,P_GS_COMPANY_ID,P_GS_BARANCH_ID,P_FROM,P_TO),0)-NVL(GS_GET_SALE_QTY_DATE_WISE(P_TYPE_ID,P_GS_COMPANY_ID,P_GS_BARANCH_ID,P_FROM,P_TO),0) INTO V_STOCK FROM DUAL;
    RETURN V_STOCK;
END;
/
create or replace function GS_GET_PURCHASE_QTY(P_TYPE_ID NUMBER,P_GS_COMPANY_ID NUMBER,	P_GS_BARANCH_ID NUMBER)
RETURN NUMBER
IS 
V_STOCK NUMBER;
BEGIN
    SELECT SUM(NVL(QUANTITY,0)) INTO V_STOCK FROM GS_GOLD_CATAGORY_TYPE GC, GS_PURCHASE_ITEM PI 
    WHERE GC.TYPE_ID = PI.TYPE_ID AND PI.GS_COMPANY_ID = P_GS_COMPANY_ID AND PI.GS_BARANCH_ID = P_GS_BARANCH_ID AND  GC.TYPE_ID = P_TYPE_ID;
    RETURN V_STOCK;
END;
/
create or replace function GS_GET_PURCHASE_QTY_DATE_WISE(P_TYPE_ID NUMBER,P_GS_COMPANY_ID NUMBER,	P_GS_BARANCH_ID NUMBER,P_FROM DATE,P_TO DATE)
RETURN NUMBER
IS 
V_STOCK NUMBER;
BEGIN
    SELECT SUM(NVL(QUANTITY,0)) INTO V_STOCK  FROM GS_GOLD_CATAGORY_TYPE GC, GS_PURCHASE GP ,GS_PURCHASE_ITEM PI 
    WHERE GC.TYPE_ID = PI.TYPE_ID 
    AND GP.PURCHASE_ID = PI.PURCHASE_ID
    AND PI.GS_COMPANY_ID = P_GS_COMPANY_ID
    AND PI.GS_BARANCH_ID = P_GS_BARANCH_ID
    AND  GC.TYPE_ID = P_TYPE_ID
    AND GP.PURCHASE_DATE BETWEEN P_FROM AND P_TO;
    RETURN V_STOCK;
END;
/
create or replace function GS_GET_SALE_QTY(P_TYPE_ID NUMBER,P_GS_COMPANY_ID NUMBER,	P_GS_BARANCH_ID NUMBER)
RETURN NUMBER
IS 
V_STOCK NUMBER;
BEGIN
    SELECT SUM(NVL(QUANTITY,0)) INTO V_STOCK FROM GS_GOLD_CATAGORY_TYPE GC, GS_SALE_ITEM SI 
    WHERE GC.TYPE_ID = SI.TYPE_ID AND SI.GS_COMPANY_ID = P_GS_COMPANY_ID AND SI.GS_BARANCH_ID = P_GS_BARANCH_ID AND GC.TYPE_ID = P_TYPE_ID;
    RETURN V_STOCK;
END;
/

IS 
V_STOCK NUMBER;
BEGIN
    SELECT SUM(NVL(QUANTITY,0)) INTO V_STOCK FROM GS_GOLD_CATAGORY_TYPE GC, GS_SALE GS, GS_SALE_ITEM SI 
    WHERE GC.TYPE_ID = SI.TYPE_ID 
    AND GS.SALE_ID = SI.SALE_ID
    AND SI.GS_COMPANY_ID = P_GS_COMPANY_ID 
    AND SI.GS_BARANCH_ID = P_GS_BARANCH_ID 
    AND GC.TYPE_ID = P_TYPE_ID
    AND GS.SALE_DATE BETWEEN P_FROM AND P_TO;
    RETURN V_STOCK;
END;
/
create or replace function GS_get_user_id_by_username(p_id varchar2)
return  NUMBER is
v_userid NUMBER;
begin
    select user_id INTO v_userid from GS_USER where UPPER(user_name) = UPPER(p_id);
    return v_userid;
end;

before insert or update ON GS_USER_ROLE 
for each row 
begin 
   if inserting then 
       if :NEW.ID is null then 
          SELECT NVL(MAX(A.ID),0)+1 INTO :NEW.ID FROM GS_USER_ROLE A ; 
       end if; 
           :NEW.CREATED_DATE    := localtimestamp; 
           :NEW.CREATED_BY := nvl(v('APP_USER'),USER); 
   end if; 
   if  updating then 
      :NEW.UPDATE_DATE    := localtimestamp; 
      :NEW.UPDATE_BY := nvl(v('APP_USER'),USER); 
   end if; 
   end;

create or replace TRIGGER "GS_USER_T" 
    BEFORE INSERT OR UPDATE 
    ON GS_USER
    FOR EACH ROW
BEGIN
    IF INSERTING THEN
        IF :NEW.USER_ID IS NULL THEN
            SELECT NVL(MAX(USER_ID), 0) + 1 INTO :NEW.USER_ID FROM GS_USER;
        END IF;
        :NEW.CREATED_DATE := SYSDATE;
        :NEW.CREATED_BY := COALESCE(SYS_CONTEXT('APEX$SESSION', 'APP_USER'), USER);
    END IF;

    :NEW.UPDATE_DATE := SYSDATE;
    :NEW.UPDATED_BY := COALESCE(SYS_CONTEXT('APEX$SESSION', 'APP_USER'), USER);
END GS_USER_T;
